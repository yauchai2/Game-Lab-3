extends SceneTree
func _initialize() -> void:call_deferred("run")
func run() -> void:
    var scene=load("res://scenes/combat/street_fight.tscn").instantiate()
    scene.preferences_path=""
    root.add_child(scene)
    scene.audio.muted=true
    scene.hud.open_page("settings")
    for i in 4:await process_frame
    scene.hud.menu_scroll.scroll_vertical=230
    for i in 4:await process_frame
    await RenderingServer.frame_post_draw
    root.get_texture().get_image().save_png("res://../docs/evidence/wev-integration/settings-credit.png")
    scene.queue_free()
    await process_frame
    call_deferred("quit")
